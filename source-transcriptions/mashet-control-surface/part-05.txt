# MASHEET GLYPH PARSER (NESTED)

import re 
 
# ---------------- 
# GLYPH MAP 
# ---------------- 
 
GLYPH_MAP = { 
    "⊕": "aggregate", 
    "⟳": "dynamics", 
    "⇢": "input", 
    "∫": "memory", 
    "⧖": "coupling", 
    "⚖": "evaluate", 
    "≋": "noise", 
} 
 
# ---------------- 
# TOKENIZER 
# ---------------- 
 
def tokenize(expr): 
    tokens = re.findall(r'[^\s{},()]+|[{},()]', expr) 
    return tokens 
 
# ---------------- 
# PARSER 
# ---------------- 
 
class Parser: 
    def __init__(self, tokens): 
        self.tokens = tokens 
        self.i = 0 
 
    def current(self): 
        return self.tokens[self.i] 
 
    def eat(self, token=None): 
        t = self.tokens[self.i] 
        if token and t != token: 
            raise SyntaxError(f"Expected {token}, got {t}") 
        self.i += 1 



        return t 
 
    def parse(self): 
        return self.parse_assignment() 
 
    def parse_assignment(self): 
        target = self.eat() 
        self.eat("←") 
        expr = self.parse_expr() 
 
        return { 
            "type": "assign", 
            "target": target, 
            "expr": expr 
        } 
 
    def parse_expr(self): 
        token = self.current() 
 
        # Case 1: aggregation 
        if token == "⊕": 
            return self.parse_aggregate() 
 
        # Case 2: function call like ∫(...) 
        if self.lookahead() == "(": 
            return self.parse_call() 
 
        # Case 3: bare glyph 
        return {"type": "glyph", "value": self.eat()} 
 
    def parse_aggregate(self): 
        self.eat("⊕") 
        self.eat("{") 
 
        args = [] 
 
        while self.current() != "}": 
            if self.current() == ",": 
                self.eat(",") 
                continue 
            args.append(self.parse_expr()) 
 
        self.eat("}") 
 



        return { 
            "type": "aggregate", 
            "args": args 
        } 
 
    def parse_call(self): 
        name = self.eat() 
        self.eat("(") 
 
        args = [] 
 
        while self.current() != ")": 
            if self.current() == ",": 
                self.eat(",") 
                continue 
            args.append(self.parse_expr()) 
 
        self.eat(")") 
 
        return { 
            "type": "call", 
            "name": name, 
            "args": args 
        } 
 
    def lookahead(self): 
        if self.i + 1 < len(self.tokens): 
            return self.tokens[self.i + 1] 
        return None
